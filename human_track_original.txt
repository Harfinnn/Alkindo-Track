class CameraState:
    """Frame, hasil deteksi, dan status satu kamera."""

    def __init__(self, camera_id, url):
        self.id = camera_id
        self.url = url
        self.lock = threading.Lock()

        # Frame terbaru dari thread capture (belum diproses AI).
        self.frame = None
        self.frame_id = 0

        # Hasil AI terakhir: (frame_id, frame, detections). Kotak digambar
        # di frame yang sama dengan yang dilihat YOLO, jadi tidak meleset.
        self.result = None

        # new_frame: capture -> inference ("ada frame baru").
        # want_frame: inference -> capture ("saya menganggur, kirim frame").
        self.new_frame = threading.Event()
        self.want_frame = threading.Event()
        self.want_frame.set()

        self.camera_ok = False
        self.last_error = ""
        self.fps = 0.0

    def get_view_data(self):
        """(frame_id, frame, detections) untuk digambar di layar."""

        with self.lock:
            if self.result is not None:
                return self.result

            return (self.frame_id, self.frame, [])

def now_string():

def write_track_log(camera_label, event, track_id, data=None, door="", direction="", transition="", duration=""):

def request_save():
    """Minta thread penulis menyimpan state (permintaan beruntun digabung)."""

    SAVE_REQUEST.set()


def append_rows(path, header, rows):
    path.parent.mkdir(parents=True, exist_ok=True)
    new_file = not path.exists()

    with open(path, "a", newline="", encoding="utf-8-sig") as file:
        writer = csv.writer(file)

        if new_file:
            writer.writerow(header)

        writer.writerows(rows)


def log_writer_worker():
    """
    Satu-satunya thread yang menulis CSV dan file state.
    Kalau file gagal ditulis (misalnya sedang dibuka di Excel), baris
    ditahan dan dicoba lagi, bukan hilang.
    """

    pending = {"occupancy": [], "track": []}
    last_warning = 0.0

    while True:
        try:
            kind, row = LOG_QUEUE.get(timeout=0.5)
            pending[kind].append(row)

            while True:
                try:
                    kind, row = LOG_QUEUE.get_nowait()
                    pending[kind].append(row)
                except queue.Empty:
                    break

        except queue.Empty:
            pass

        for kind, rows in pending.items():
            if not rows:
                continue

            path, header = CSV_TARGETS[kind]

            try:
                append_rows(path, header, rows)
                rows.clear()

            except Exception as error:
                if len(rows) > MAX_PENDING_ROWS:
                    del rows[: len(rows) - MAX_PENDING_ROWS]

                if time.monotonic() - last_warning > 10.0:
                    last_warning = time.monotonic()
                    print(
                        f"[WARNING] Gagal menulis {path} ({error}). "
                        f"{len(rows)} baris ditahan, dicoba lagi."
                    )

        if SAVE_REQUEST.is_set():
            SAVE_REQUEST.clear()
            save_state()

        if LOG_STOP.is_set() and LOG_QUEUE.empty():
            break


def print_track_summary(cam, stamp):
    camera_id = cam.id

    if stamp - LAST_TRACK_SUMMARY[camera_id] < TRACK_SUMMARY_INTERVAL:
        return

    LAST_TRACK_SUMMARY[camera_id] = stamp

    stats = TRACK_STATS[camera_id]
    area_counts = {name: 0 for name in AREA_NAMES}
    active_total = 0

    for data in list(TRACKED_PEOPLE[camera_id].values()):
        if stamp - data.get("last_seen", 0) <= TRACK_ACTIVE_WINDOW:
            active_total += 1
            area = data.get("area", "BELUM DIKETAHUI")
            area_counts[area if area in area_counts else "BELUM DIKETAHUI"] += 1

    print(
        f"\n[TRACK SUMMARY {CAMERAS[camera_id]['label']}] "
        f"{datetime.now().strftime('%H:%M:%S')} | "
        f"ACTIVE={active_total} | NEW={stats['new']} | "
        f"LOST={stats['lost']} | CROSSING={stats['crossing']} | FPS={cam.fps:.1f}"
    )
    for area in AREA_NAMES:
        print(f"  {area:<16}: {area_counts[area]} active")
    print("")

def save_config():
    data = {"doors": {}}

    for door_id, door in DOORS.items():
        line = door["line"]
        normalized_line = None

        frame_width, frame_height = FRAME_SIZES[door["camera"]]

        if line is not None:
            normalized_line = [
                [x / frame_width, y / frame_height]
                for x, y in line
            ]

        data["doors"][str(door_id)] = {
            "camera": door["camera"],
            "line": normalized_line,
            "reverse": door["reverse"],
        }

    CONFIG_FILE.parent.mkdir(parents=True, exist_ok=True)
    with open(CONFIG_FILE, "w", encoding="utf-8") as file:
        json.dump(data, file, indent=4)

    print("[CONFIG] Konfigurasi garis tersimpan.")


def load_config():
    if not CONFIG_FILE.exists():
        print("[CONFIG] Belum ada konfigurasi garis.")
        return

    try:
        with open(CONFIG_FILE, "r", encoding="utf-8") as file:
            data = json.load(file)

        for key, config in data.get("doors", {}).items():
            door_id = int(key)

            if door_id not in DOORS:
                continue

            door = DOORS[door_id]

            # Konfigurasi lama (satu kamera) tidak punya field "camera"
            # dan semua garisnya milik kamera lobby.
            saved_camera = config.get("camera", "LANTAI1")
            saved_camera = CAMERA_ALIASES.get(saved_camera, saved_camera)

            if saved_camera != door["camera"]:
                print(
                    f"[CONFIG] Garis {door['name']} lama berasal dari kamera "
                    f"{saved_camera}, diabaikan. Pasang ulang di kamera "
                    f"{CAMERAS[door['camera']]['title']} (tekan {door_id})."
                )
                continue

            frame_width, frame_height = FRAME_SIZES[door["camera"]]
            normalized_line = config.get("line")

            if normalized_line:
                door["line"] = [
                    (
                        int(point[0] * frame_width),
                        int(point[1] * frame_height),
                    )
                    for point in normalized_line
                ]

            door["reverse"] = config.get("reverse", False)

        print("[CONFIG] Konfigurasi berhasil dimuat.")

    except Exception as error:
        print("[ERROR] Gagal memuat konfigurasi:", error)

def save_state():
    """Simpan okupansi dan hitungan IN/OUT supaya selamat dari restart."""

    # Snapshot cepat di bawah lock; tulis file di luar lock.
    with STATE_LOCK:
        data = {
            "date": datetime.now().strftime("%Y-%m-%d"),
            "occupancy": dict(AREA_OCCUPANCY),
            "counts": {
                str(door_id): dict(value)
                for door_id, value in COUNTS.items()
            },
        }

    with STATE_IO_LOCK:
        STATE_FILE.parent.mkdir(parents=True, exist_ok=True)
        tmp_file = STATE_FILE.with_suffix(".tmp")

        try:
            with open(tmp_file, "w", encoding="utf-8") as file:
                json.dump(data, file, indent=2)

            # Atomik: file lama tidak rusak kalau mati listrik.
            tmp_file.replace(STATE_FILE)

        except Exception as error:
            print("[ERROR] Gagal menyimpan state:", error)


def load_state():
    """Pulihkan state hanya kalau tanggalnya hari ini."""

    if not STATE_FILE.exists():
        print("[STATE] Belum ada state tersimpan.")
        return

    try:
        with open(STATE_FILE, "r", encoding="utf-8") as file:
            data = json.load(file)

        today = datetime.now().strftime("%Y-%m-%d")

        if data.get("date") != today:
            print("[STATE] State hari lain diabaikan, mulai dari 0.")
            return

        saved_occupancy = data.get("occupancy", {})

        # State lama memakai nama "Ruang IT" untuk area yang sama.
        if "Office Lantai 1" not in saved_occupancy and "Ruang IT" in saved_occupancy:
            saved_occupancy["Office Lantai 1"] = saved_occupancy["Ruang IT"]

        with STATE_LOCK:
            for key in AREA_OCCUPANCY:
                AREA_OCCUPANCY[key] = max(0, int(saved_occupancy.get(key, 0)))

            for key, value in data.get("counts", {}).items():
                door_id = int(key)

                if door_id in COUNTS:
                    COUNTS[door_id]["IN"] = int(value.get("IN", 0))
                    COUNTS[door_id]["OUT"] = int(value.get("OUT", 0))

        print(f"[STATE] Dipulihkan: {AREA_OCCUPANCY}")

    except Exception as error:
        print("[ERROR] Gagal memuat state:", error)


def reset_counts(reason):
    """Reset semua hitungan crossing dan okupansi area."""

    with STATE_LOCK:
        for door_id in DOORS:
            COUNTS[door_id]["IN"] = 0
            COUNTS[door_id]["OUT"] = 0

            TRACK_SIDES[door_id].clear()
            LAST_FOOT_POINT[door_id].clear()
            LAST_COUNT_TIME[door_id].clear()

        for area_name in AREA_OCCUPANCY:
            AREA_OCCUPANCY[area_name] = 0

    request_save()

    print(f"[INFO] Hitungan dan okupansi direset ({reason}).")


def adjust_area(area_key, delta):
    """Koreksi manual okupansi satu area."""

    with STATE_LOCK:
        AREA_OCCUPANCY[area_key] = max(0, AREA_OCCUPANCY[area_key] + delta)
        new_value = AREA_OCCUPANCY[area_key]

    request_save()

    print(f"[KOREKSI] {area_key} {delta:+d} -> {new_value}")


def set_area(area_key, value):
    """Isi okupansi satu area secara manual."""

    with STATE_LOCK:
        AREA_OCCUPANCY[area_key] = max(0, int(value))
        new_value = AREA_OCCUPANCY[area_key]

    request_save()

    print(f"[KOREKSI] {area_key} diatur ke {new_value}")


def get_snapshot():
    """Salinan okupansi dan hitungan untuk digambar di panel."""

    with STATE_LOCK:
        occupancy = dict(AREA_OCCUPANCY)
        counts = {
            door_id: dict(value)
            for door_id, value in COUNTS.items()
        }

    return occupancy, counts


def make_state_key(snapshot):
    """
    Kunci murah yang berubah kalau okupansi, hitungan, atau garis pintu
    berubah. Dipakai untuk memutuskan perlu menggambar ulang panel atau tidak.
    """

    occupancy, counts = snapshot

    lines = tuple(
        (
            tuple(door["line"]) if door["line"] else None,
            door["reverse"],
        )
        for door in DOORS.values()
    )

    return (
        tuple(sorted(occupancy.items())),
        tuple(
            (door_id, value["IN"], value["OUT"])
            for door_id, value in sorted(counts.items())
        ),
        lines,
    )


# ============================================================
# 5. GEOMETRI GARIS
# ============================================================


def calculate_side(point, line_start, line_end):
    """Jarak bertanda (piksel) titik terhadap garis."""

    x, y = point
    x1, y1 = line_start
    x2, y2 = line_end

    length = math.hypot(x2 - x1, y2 - y1)

    if length < 1:
        return 0.0

    return ((x2 - x1) * (y - y1) - (y2 - y1) * (x - x1)) / length


def calculate_crossing_point(
    previous_point,
    current_point,
    line_start,
    line_end,
):
    """Hitung titik perpotongan lintasan dengan garis virtual."""

    previous_value = calculate_side(previous_point, line_start, line_end)
    current_value = calculate_side(current_point, line_start, line_end)

    denominator = previous_value - current_value

    if abs(denominator) < 1e-6:
        return None

    ratio = previous_value / denominator

    x = previous_point[0] + ratio * (current_point[0] - previous_point[0])
    y = previous_point[1] + ratio * (current_point[1] - previous_point[1])

    return x, y


def point_on_segment(point, line_start, line_end):
    """Pastikan titik crossing berada pada segmen garis pintu."""

    px, py = point
    x1, y1 = line_start
    x2, y2 = line_end

    dx = x2 - x1
    dy = y2 - y1

    length_squared = dx * dx + dy * dy

    if length_squared == 0:
        return False

    ratio = ((px - x1) * dx + (py - y1) * dy) / length_squared

    if ratio < 0 or ratio > 1:
        return False

    projected_x = x1 + ratio * dx
    projected_y = y1 + ratio * dy

    distance = math.hypot(px - projected_x, py - projected_y)

    return distance <= 30

def decrement(area_key):
    """Kurangi okupansi area; beri peringatan kalau sudah 0."""

    if AREA_OCCUPANCY[area_key] <= 0:
        print(
            f"[WARNING] {area_key} sudah 0 tapi ada orang keluar "
            "(hitungan meleset / ID switch / orang sudah ada sebelum start)."
        )
        return

    AREA_OCCUPANCY[area_key] -= 1


def update_area_occupancy(door_id, direction):
    """
    Kamera LANTAI1:
    Pintu 1:  IN = luar gedung -> Lobby      | OUT = Lobby -> luar gedung
    Pintu 2:  IN = Lobby -> Area Produksi    | OUT = Area Produksi -> Lobby
    Pintu 3:  IN = Lobby -> Office Lantai 1  | OUT = Office Lantai 1 -> Lobby

    Kamera LANTAI2:
    Pintu 4 (tangga):  IN = naik, Lobby -> Lantai 2
                       OUT = turun, Lantai 2 -> Lobby
    Pintu 5 (office):  IN = Lantai 2 -> Office Lantai 2
                       OUT = Office Lantai 2 -> Lantai 2
    Pintu 6 (meeting): IN = Lantai 2 -> Ruang Meeting 2
                       OUT = Ruang Meeting 2 -> Lantai 2
    """

    transition = None

    if door_id == 1:
        if direction == "IN":
            AREA_OCCUPANCY["Lobby"] += 1
            transition = "Luar -> Lobby"
        else:
            decrement("Lobby")
            transition = "Lobby -> Luar"

    elif door_id == 2:
        if direction == "IN":
            decrement("Lobby")
            AREA_OCCUPANCY["Area Produksi"] += 1
            transition = "Lobby -> Area Produksi"
        else:
            decrement("Area Produksi")
            AREA_OCCUPANCY["Lobby"] += 1
            transition = "Area Produksi -> Lobby"

    elif door_id == 3:
        if direction == "IN":
            decrement("Lobby")
            AREA_OCCUPANCY["Office Lantai 1"] += 1
            transition = "Lobby -> Office Lantai 1"
        else:
            decrement("Office Lantai 1")
            AREA_OCCUPANCY["Lobby"] += 1
            transition = "Office Lantai 1 -> Lobby"

    elif door_id == 4:
        if direction == "IN":
            decrement("Lobby")
            AREA_OCCUPANCY["Lantai 2"] += 1
            transition = "Lobby -> Lantai 2"
        else:
            decrement("Lantai 2")
            AREA_OCCUPANCY["Lobby"] += 1
            transition = "Lantai 2 -> Lobby"

    elif door_id == 5:
        if direction == "IN":
            decrement("Lantai 2")
            AREA_OCCUPANCY["Office Lantai 2"] += 1
            transition = "Lantai 2 -> Office Lantai 2"
        else:
            decrement("Office Lantai 2")
            AREA_OCCUPANCY["Lantai 2"] += 1
            transition = "Office Lantai 2 -> Lantai 2"

    elif door_id == 6:
        if direction == "IN":
            decrement("Lantai 2")
            AREA_OCCUPANCY["Ruang Meeting 2"] += 1
            transition = "Lantai 2 -> Ruang Meeting 2"
        else:
            decrement("Ruang Meeting 2")
            AREA_OCCUPANCY["Lantai 2"] += 1
            transition = "Ruang Meeting 2 -> Lantai 2"

    elif door_id == 7:
        if direction == "IN":
            decrement("Lantai 2")
            AREA_OCCUPANCY["Ruang Meeting 1"] += 1
            transition = "Lantai 2 -> Ruang Meeting 1"
        else:
            decrement("Ruang Meeting 1")
            AREA_OCCUPANCY["Lantai 2"] += 1
            transition = "Ruang Meeting 1 -> Lantai 2"

    return transition

def _process_crossing(door_id, track_id, foot_point):
    """
    Logika crossing. Harus dipanggil di bawah STATE_LOCK.
    Mengembalikan data event kalau ada hitungan baru, selain itu None.
    """

    door = DOORS[door_id]
    line = door["line"]

    if line is None:
        return

    line_start, line_end = line

    # Margin dalam piksel, mengikuti resolusi frame kamera pintu ini.
    frame_width = FRAME_SIZES[door["camera"]][0]
    margin = CROSSING_MARGIN_RATIO * frame_width

    previous_point = LAST_FOOT_POINT[door_id].get(track_id)

    if previous_point is None:
        LAST_FOOT_POINT[door_id][track_id] = foot_point

        side_value = calculate_side(foot_point, line_start, line_end)

        if abs(side_value) >= margin:
            TRACK_SIDES[door_id][track_id] = 1 if side_value > 0 else -1

        return

    previous_side = TRACK_SIDES[door_id].get(track_id)

    current_value = calculate_side(foot_point, line_start, line_end)

    LAST_FOOT_POINT[door_id][track_id] = foot_point

    if abs(current_value) < margin:
        return

    current_side = 1 if current_value > 0 else -1

    if previous_side is None:
        TRACK_SIDES[door_id][track_id] = current_side
        return

    if previous_side == current_side:
        return

    crossing_point = calculate_crossing_point(
        previous_point,
        foot_point,
        line_start,
        line_end,
    )

    TRACK_SIDES[door_id][track_id] = current_side

    if crossing_point is None:
        return

    if not point_on_segment(crossing_point, line_start, line_end):
        return

    now = time.monotonic()
    last_time = LAST_COUNT_TIME[door_id].get(track_id, 0)

    if now - last_time < COUNT_COOLDOWN:
        return

    # Arah default:
    # sisi negatif -> positif = IN
    # sisi positif -> negatif = OUT
    if not door["reverse"]:
        direction = "IN" if previous_side == -1 else "OUT"
    else:
        direction = "OUT" if previous_side == -1 else "IN"

    COUNTS[door_id][direction] += 1

    transition = update_area_occupancy(door_id, direction)

    LAST_COUNT_TIME[door_id][track_id] = now

    return door["name"], direction, track_id, dict(AREA_OCCUPANCY), transition


def process_crossing(door_id, track_id, foot_point):
    """Hitung crossing di bawah lock, lalu catat log/state di luar lock."""

    with STATE_LOCK:
        event = _process_crossing(door_id, track_id, foot_point)

    if event is None:
        return

    door_name, direction, event_track_id, occupancy, transition = event

    camera_id = DOORS[door_id]["camera"]
    camera_label = CAMERAS[camera_id]["label"]

    destination = transition.split(" -> ")[-1] if transition else "BELUM DIKETAHUI"

    people = TRACKED_PEOPLE[camera_id]
    track_data = people.get(event_track_id, {})
    track_data["area"] = normalize_area_name(destination)
    people[event_track_id] = track_data

    TRACK_STATS[camera_id]["crossing"] += 1

    print(
        f"[CROSSING] {datetime.now().strftime('%H:%M:%S')} | "
        f"{camera_label} | "
        f"ID={event_track_id} | {normalize_area_name(transition.split(' -> ')[0]) if transition else 'AREA?'} "
        f"-> {normalize_area_name(destination)} | {direction} | PINTU={door_name}"
    )
    write_track_log(camera_label, "CROSSING", event_track_id, track_data, door_name, direction, transition)
    write_log(camera_label, door_name, direction, event_track_id, occupancy)

    # Event crossing jarang terjadi, jadi aman disimpan tiap kali
    # (ditulis thread khusus, bukan thread YOLO).
    request_save()

def put_text_right(frame, text, x_right, y, scale, color, thickness=1):
    """Teks rata kanan, supaya angka sejajar."""

    (tw, _), _ = cv2.getTextSize(
        text, cv2.FONT_HERSHEY_SIMPLEX, scale, thickness
    )

    cv2.putText(
        frame,
        text,
        (x_right - tw, y),
        cv2.FONT_HERSHEY_SIMPLEX,
        scale,
        color,
        thickness,
        cv2.LINE_AA,
    )


def draw_detections(frame, detections):
    """Tempel kotak dan ID hasil deteksi ke frame."""

    for det in detections:
        x1, y1, x2, y2 = det["box"]

        cv2.rectangle(frame, (x1, y1), (x2, y2), (0, 255, 0), 1)

        cv2.putText(
            frame,
            f"PERSON ID {det['id']}",
            (x1, max(16, y1 - 5)),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.4,
            (0, 255, 0),
            1,
            cv2.LINE_AA,
        )

        cv2.circle(frame, det["foot"], 3, (0, 0, 255), -1)


def draw_door_lines(frame, camera_id):
    """Garis virtual pintu milik kamera ini beserta label arahnya."""

    h, w = frame.shape[:2]
    font = cv2.FONT_HERSHEY_SIMPLEX

    for door_id in doors_of_camera(camera_id):
        door = DOORS[door_id]
        line = door["line"]
        color = door["color"]

        if line is None:
            continue

        x1, y1 = line[0]
        x2, y2 = line[1]

        cv2.line(frame, (x1, y1), (x2, y2), color, 2)
        cv2.circle(frame, (x1, y1), 4, color, -1)
        cv2.circle(frame, (x2, y2), 4, color, -1)

        middle_x = (x1 + x2) // 2
        middle_y = (y1 + y2) // 2

        # Nama pintu
        cv2.putText(
            frame,
            door["name"],
            (max(5, middle_x - 60), max(20, middle_y - 10)),
            font,
            0.38,
            color,
            1,
            cv2.LINE_AA,
        )

        # Arah tegak lurus garis
        dx = x2 - x1
        dy = y2 - y1
        length = math.hypot(dx, dy)

        if length < 1:
            continue

        nx = -dy / length
        ny = dx / length
        offset = 32

        if door.get("reverse", False):
            neg_text, pos_text = door["label_pos"], door["label_neg"]
        else:
            neg_text, pos_text = door["label_neg"], door["label_pos"]

        labels = [
            (
                int(middle_x - nx * offset),
                int(middle_y - ny * offset),
                neg_text,
                (0, 0, 180),
            ),
            (
                int(middle_x + nx * offset),
                int(middle_y + ny * offset),
                pos_text,
                (0, 130, 0),
            ),
        ]

        for lx, ly, text, bg_color in labels:
            font_scale = 0.3
            thickness = 1

            (tw, th), _ = cv2.getTextSize(text, font, font_scale, thickness)

            lx = max(5, min(lx - tw // 2, w - tw - 10))
            ly = max(th + 8, min(ly, h - 8))

            cv2.rectangle(
                frame,
                (lx - 5, ly - th - 5),
                (lx + tw + 5, ly + 5),
                bg_color,
                -1,
            )

            cv2.putText(
                frame,
                text,
                (lx, ly),
                font,
                font_scale,
                (255, 255, 255),
                thickness,
                cv2.LINE_AA,
            )


def draw_info_panel(frame, fps_text, snapshot):
    """Panel kecil: jumlah orang per area + IN/OUT pintu terkait."""

    h, w = frame.shape[:2]
    font = cv2.FONT_HERSHEY_SIMPLEX

    occupancy, counts = snapshot

    row_h = 14
    panel_w = 150
    panel_h = 22 + len(PANEL_AREAS) * row_h + 12
    x = max(0, w - panel_w - 6)
    y = 6

    # Latar transparan hanya di area panel (tanpa copy seluruh frame).
    roi = frame[y:y + panel_h, x:x + panel_w]
    background = np.full_like(roi, (20, 20, 20))
    cv2.addWeighted(background, 0.75, roi, 0.25, 0, roi)

    col_orang = x + 90
    col_in = x + 116
    col_out = x + 144
    gray = (170, 170, 170)

    # Judul kolom
    cv2.putText(
        frame, "AREA", (x + 11, y + 12), font, 0.3, gray, 1, cv2.LINE_AA
    )
    put_text_right(frame, "ORANG", col_orang, y + 12, 0.3, gray)
    put_text_right(frame, "IN", col_in, y + 12, 0.3, (0, 255, 100))
    put_text_right(frame, "OUT", col_out, y + 12, 0.3, (0, 100, 255))

    cv2.line(
        frame,
        (x + 5, y + 17),
        (x + panel_w - 5, y + 17),
        (90, 90, 90),
        1,
    )

    for i, (label, area_key, door_id) in enumerate(PANEL_AREAS):
        top = y + 20 + i * row_h
        base = top + 10

        # Garis warna = warna garis pintu di video.
        cv2.rectangle(
            frame,
            (x + 5, top + 1),
            (x + 7, top + row_h - 2),
            DOORS[door_id]["color"],
            -1,
        )

        cv2.putText(
            frame, label, (x + 11, base), font, 0.35,
            (255, 255, 255), 1, cv2.LINE_AA,
        )

        put_text_right(
            frame, str(occupancy[area_key]),
            col_orang, base, 0.45, (255, 255, 255), 1,
        )

        put_text_right(
            frame, str(counts[door_id]["IN"]),
            col_in, base, 0.35, (0, 255, 100),
        )

        put_text_right(
            frame, str(counts[door_id]["OUT"]),
            col_out, base, 0.35, (0, 100, 255),
        )

    cv2.putText(
        frame, f"FPS {fps_text}", (x + 6, y + panel_h - 4),
        font, 0.28, gray, 1, cv2.LINE_AA,
    )

def render_camera(cam, view_data, fps_text, snapshot):
    """Gambar satu panel kamera (ukuran asli frame kamera)."""

    camera_id = cam.id
    _, raw_frame, detections = view_data

    if raw_frame is None:
        frame_width, frame_height = FRAME_SIZES[camera_id]
        frame = np.zeros((frame_height, frame_width, 3), dtype="uint8")

        cv2.putText(
            frame,
            "MENUNGGU CCTV...",
            (30, 50),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.7,
            (0, 200, 255),
            1,
        )

    else:
        # Frame bersama tidak boleh digambari, jadi di-copy
        # (hanya saat panel memang perlu digambar ulang).
        frame = raw_frame.copy()

        draw_detections(frame, detections)
        draw_door_lines(frame, camera_id)

        if camera_id == INFO_PANEL_CAMERA:
            draw_info_panel(frame, fps_text, snapshot)

        if SELECTING_LINE and DOORS[ACTIVE_DOOR]["camera"] == camera_id:
            message = (
                f"SET GARIS: "
                f"{DOORS[ACTIVE_DOOR]['name']} - "
                f"KLIK {len(PENDING_POINTS) + 1}/2"
            )

            cv2.putText(
                frame,
                message,
                (8, 36),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.5,
                DOORS[ACTIVE_DOOR]["color"],
                1,
                cv2.LINE_AA,
            )

            for point in list(PENDING_POINTS):
                cv2.circle(
                    frame,
                    point,
                    5,
                    DOORS[ACTIVE_DOOR]["color"],
                    -1,
                )

    # Judul kamera
    cv2.putText(
        frame,
        f"{CAMERAS[camera_id]['title']} | FPS {cam.fps:.0f}",
        (8, 16),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.4,
        (255, 255, 255),
        1,
        cv2.LINE_AA,
    )

    if not cam.camera_ok:
        cv2.putText(
            frame,
            "CAMERA RECONNECTING",
            (8, 56),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.5,
            (0, 0, 255),
            1,
            cv2.LINE_AA,
        )

    elif cam.last_error:
        cv2.putText(
            frame,
            cam.last_error[:70],
            (8, 56),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.4,
            (0, 0, 255),
            1,
            cv2.LINE_AA,
        )

    return frame

def mouse_callback(event, x, y, flags, param):
    global PENDING_POINTS
    global SELECTING_LINE

    if event != cv2.EVENT_LBUTTONDOWN:
        return

    if not SELECTING_LINE:
        return

    # Garis hanya boleh dipasang di panel kamera milik pintu aktif.
    camera_id = DOORS[ACTIVE_DOOR]["camera"]
    view = VIEWS.get(camera_id)

    if view is None:
        return

    local_x = x - view["x"]
    local_y = y - view["y"]

    if not (0 <= local_x < view["w"] and 0 <= local_y < view["h"]):
        print(
            f"[WARNING] {DOORS[ACTIVE_DOOR]['name']} dipasang di panel "
            f"{CAMERAS[camera_id]['title']}. Klik di panel itu."
        )
        return

    frame_width, frame_height = FRAME_SIZES[camera_id]

    original_x = int(local_x * frame_width / view["w"])
    original_y = int(local_y * frame_height / view["h"])

    original_x = max(0, min(frame_width - 1, original_x))
    original_y = max(0, min(frame_height - 1, original_y))

    PENDING_POINTS.append((original_x, original_y))

    print(
        f"[{DOORS[ACTIVE_DOOR]['name']}] "
        f"Titik {len(PENDING_POINTS)}/2: "
        f"{original_x}, {original_y}"
    )

    if len(PENDING_POINTS) < 2:
        return

    if PENDING_POINTS[0] == PENDING_POINTS[1]:
        print("[WARNING] Kedua titik tidak boleh sama.")
        PENDING_POINTS = []
        return

    with STATE_LOCK:
        DOORS[ACTIVE_DOOR]["line"] = [
            PENDING_POINTS[0],
            PENDING_POINTS[1],
        ]

        TRACK_SIDES[ACTIVE_DOOR].clear()
        LAST_FOOT_POINT[ACTIVE_DOOR].clear()
        LAST_COUNT_TIME[ACTIVE_DOOR].clear()

    PENDING_POINTS = []
    SELECTING_LINE = False

    try:
        save_config()
        print(f"[OK] Garis {DOORS[ACTIVE_DOOR]['name']} berhasil dipasang.")

    except Exception as error:
        print("[ERROR] Gagal menyimpan garis:", error)

def load_model(device_hint=None, use_openvino=USE_OPENVINO):
    """
    Muat satu instance model YOLO. Prioritas:
    1. OpenVINO (export otomatis sekali kalau belum ada)
    2. Model .pt biasa (fallback kalau OpenVINO gagal)

    Tiap kamera memakai instance sendiri supaya tracker (persist=True)
    tidak tercampur antar kamera. device_hint dipakai instance kedua
    supaya tes iGPU tidak diulang.
    """

    model_path = Path(MODEL_PATH)

    if use_openvino and model_path.suffix == ".pt":
        openvino_dir = model_path.with_name(
            f"{model_path.stem}_openvino_model"
        )

        try:
            if not openvino_dir.exists():
                print(
                    "[MODEL] Export OpenVINO pertama kali "
                    "(sekali saja, bisa 1-2 menit)..."
                )

                YOLO(MODEL_PATH).export(
                    format="openvino",
                    imgsz=IMGSZ,
                    half=False,
                )

            device = device_hint or INFERENCE_DEVICE

            if device == "intel:gpu" and device_hint is None:
                # Tes iGPU dengan instance terpisah (dibuang setelah tes),
                # supaya model utama / tracker tidak terpengaruh.
                try:
                    probe = YOLO(str(openvino_dir), task="detect")

                    probe.predict(
                        np.zeros((IMGSZ, IMGSZ, 3), dtype="uint8"),
                        imgsz=IMGSZ,
                        device=device,
                        verbose=False,
                    )

                    del probe

                except Exception as error:
                    print("[MODEL] iGPU tidak bisa dipakai, pakai CPU:", error)
                    device = "intel:cpu"

            model = YOLO(str(openvino_dir), task="detect")

            print(f"[MODEL] Backend OpenVINO ({device}): {openvino_dir}")

            return model, device

        except Exception as error:
            print("[MODEL] OpenVINO gagal, pakai .pt:", error)
            print("[MODEL] Pasang dengan: pip install openvino")

    print(f"[MODEL] Backend PyTorch: {MODEL_PATH}")

    return YOLO(MODEL_PATH), None


def open_capture(url):
    """Buka RTSP dengan batas waktu supaya tidak menggantung."""

    try:
        cap = cv2.VideoCapture(
            url,
            cv2.CAP_FFMPEG,
            [
                cv2.CAP_PROP_OPEN_TIMEOUT_MSEC, CAPTURE_TIMEOUT_MS,
                cv2.CAP_PROP_READ_TIMEOUT_MSEC, CAPTURE_TIMEOUT_MS,
            ],
        )

    except Exception:
        # OpenCV lama tanpa parameter timeout.
        cap = cv2.VideoCapture(url, cv2.CAP_FFMPEG)

    cap.set(cv2.CAP_PROP_BUFFERSIZE, 1)

    return cap


def capture_worker(cam):
    """
    Thread penerima frame satu kamera.

    Frame selalu ditarik (grab) supaya buffer RTSP terkuras dan gambar
    tidak tertinggal, tapi konversi warna + resize (retrieve) hanya
    dilakukan kalau YOLO sedang menganggur dan jeda minimum sudah lewat.
    """

    cap = None

    publish_interval = 1.0 / max(1, INFERENCE_MAX_FPS)
    next_publish = 0.0

    while not STOP_EVENT.is_set():
        try:
            if cap is None or not cap.isOpened():
                if cap is not None:
                    cap.release()

                cap = open_capture(cam.url)

                if not cap.isOpened():
                    cam.camera_ok = False
                    cam.last_error = "RTSP belum tersambung"

                    time.sleep(RECONNECT_DELAY)
                    continue

            if not cap.grab():
                ok, frame = False, None

            elif (
                not cam.want_frame.is_set()
                or time.monotonic() < next_publish
            ):
                # YOLO masih sibuk atau terlalu cepat: buang frame ini
                # tanpa konversi warna / resize.
                cam.camera_ok = True
                cam.last_error = ""
                continue

            else:
                ok, frame = cap.retrieve()

                now = time.monotonic()
                next_publish += publish_interval

                # Kalau tertinggal (YOLO lebih lambat dari batas FPS atau
                # setelah reconnect), jangan mengejar.
                if next_publish < now:
                    next_publish = now

            if not ok or frame is None or frame.size == 0:
                cam.camera_ok = False
                cam.last_error = "Frame CCTV kosong; mencoba reconnect"

                if cap is not None:
                    cap.release()

                cap = None
                time.sleep(RECONNECT_DELAY)
                continue

            cam.camera_ok = True
            cam.last_error = ""

            # Kecilkan sekali di sini. Semua bagian lain
            # (YOLO, kotak, dashboard) memakai versi kecil ini.
            h, w = frame.shape[:2]
            scale = min(DISPLAY_WIDTH / w, DISPLAY_HEIGHT / h)

            if scale < 1.0:
                frame = cv2.resize(
                    frame,
                    (int(w * scale), int(h * scale)),
                    interpolation=cv2.INTER_AREA,
                )

            with cam.lock:
                cam.frame = frame
                cam.frame_id += 1

            # Satu frame dalam proses sekali waktu.
            cam.want_frame.clear()
            cam.new_frame.set()

        except Exception as error:
            cam.camera_ok = False
            cam.last_error = str(error)

            if cap is not None:
                cap.release()
                cap = None

            time.sleep(RECONNECT_DELAY)

    if cap is not None:
        cap.release()


def inference_worker(cam, model, inference_device):
    """Thread YOLO + ByteTrack satu kamera."""

    camera_id = cam.id
    camera_label = CAMERAS[camera_id]["label"]
    people = TRACKED_PEOPLE[camera_id]
    stats = TRACK_STATS[camera_id]
    door_ids = doors_of_camera(camera_id)

    previous_time = time.monotonic()

    while not STOP_EVENT.is_set():
        # Tunggu frame baru tanpa polling.
        if not cam.new_frame.wait(timeout=0.5):
            continue

        cam.new_frame.clear()

        # Frame hanya dibaca (tidak dimodifikasi), jadi tidak perlu copy.
        with cam.lock:
            frame = cam.frame
            frame_id = cam.frame_id

        if frame is None:
            cam.want_frame.set()
            continue

        detections = []

        try:
            results = model.track(
                frame,
                persist=True,
                classes=[PERSON_CLASS_ID],
                conf=CONFIDENCE,
                imgsz=IMGSZ,
                tracker=TRACKER_CONFIG,
                device=inference_device,
                verbose=False,
            )

            if results:
                result = results[0]

                if result.boxes is not None and result.boxes.id is not None:
                    boxes = result.boxes.xyxy.cpu().numpy()
                    track_ids = result.boxes.id.int().cpu().tolist()
                    confidences = (
                        result.boxes.conf.cpu().numpy().tolist()
                        if result.boxes.conf is not None
                        else [None] * len(track_ids)
                    )

                    stamp = time.monotonic()

                    for box, track_id, confidence in zip(boxes, track_ids, confidences):
                        x1, y1, x2, y2 = map(int, box)

                        # Titik kaki menjadi titik referensi.
                        foot_point = ((x1 + x2) // 2, y2)
                        is_new_track = track_id not in people
                        previous = people.get(track_id, {})

                        people[track_id] = {
                            "first_seen": previous.get("first_seen", stamp),
                            "last_seen": stamp,
                            "point": foot_point,
                            "box": (x1, y1, x2, y2),
                            "confidence": round(float(confidence), 4) if confidence is not None else "",
                            "area": previous.get("area", "BELUM DIKETAHUI"),
                        }

                        if is_new_track:
                            stats["new"] += 1

                            # Simpan NEW TRACK ke CSV saja.
                            # Tidak perlu mencetak ke terminal.
                            write_track_log(
                                camera_label,
                                "NEW_TRACK",
                                track_id,
                                people[track_id],
                            )

                        # Periksa garis virtual milik kamera ini saja.
                        for door_id in door_ids:
                            process_crossing(
                                door_id,
                                track_id,
                                foot_point,
                            )

                        detections.append(
                            {
                                "box": (x1, y1, x2, y2),
                                "id": track_id,
                                "foot": foot_point,
                            }
                        )

            # Bersihkan state ID yang sudah lama hilang.
            stamp = time.monotonic()

            expired_ids = [
                track_id
                for track_id, data in people.items()
                if stamp - data["last_seen"] > MAX_TRACK_LOST_SECONDS
            ]

            if expired_ids:
                removed = []

                with STATE_LOCK:
                    for track_id in expired_ids:
                        lost_data = people.pop(track_id, None) or {}
                        stats["lost"] += 1
                        removed.append((track_id, lost_data))

                        for door_id in door_ids:
                            TRACK_SIDES[door_id].pop(track_id, None)
                            LAST_FOOT_POINT[door_id].pop(track_id, None)
                            LAST_COUNT_TIME[door_id].pop(track_id, None)

                # Print dan catat CSV di luar lock.
                for track_id, lost_data in removed:
                    duration = max(0.0, stamp - lost_data.get("first_seen", stamp))

                    print(
                        f"[TRACK LOST] {datetime.now().strftime('%H:%M:%S')} | "
                        f"{camera_label} | ID={track_id} | "
                        f"LAST AREA={lost_data.get('area', 'BELUM DIKETAHUI')} | "
                        f"DURATION={duration:.1f}s"
                    )
                    write_track_log(
                        camera_label,
                        "TRACK_LOST",
                        track_id,
                        lost_data,
                        duration=round(duration, 1),
                    )

            print_track_summary(cam, stamp)

            current_time = time.monotonic()
            elapsed = current_time - previous_time

            if elapsed > 0:
                cam.fps = 0.9 * cam.fps + 0.1 * (1.0 / elapsed)

            previous_time = current_time

            # Simpan frame + hasilnya sebagai satu pasangan (daftar kosong
            # kalau tidak ada orang, supaya kotak lama tidak tertinggal).
            with cam.lock:
                cam.result = (frame_id, frame, detections)

        except Exception as error:
            cam.last_error = f"Inferensi YOLO: {error}"

            print("[ERROR]", cam.last_error)

            with cam.lock:
                cam.result = (frame_id, frame, [])

            time.sleep(0.2)

        finally:
            # Beri tahu capture: siap menerima frame berikutnya.
            cam.want_frame.set()



def load_camera_urls():
    """Membaca konfigurasi CCTV Office dari cameras.json."""

    if not CAMERA_CONFIG_FILE.exists():
        raise FileNotFoundError(
            f"File konfigurasi CCTV tidak ditemukan: {CAMERA_CONFIG_FILE}"
        )

    with open(CAMERA_CONFIG_FILE, "r", encoding="utf-8") as file:
        config = json.load(file)

    # Ambil kelompok OFFICE.
    office_config = config.get("OFFICE", config)

    urls = {}

    for camera_id in CAMERAS:
        if camera_id not in office_config:
            raise ValueError(
                f"Konfigurasi OFFICE.{camera_id} "
                f"tidak ditemukan di cameras.json"
            )

        camera = office_config[camera_id]

        ip = camera["ip"]
        username = quote(str(camera["username"]), safe="")
        password = quote(str(camera["password"]), safe="")
        port = int(camera.get("port", 554))
        channel = camera.get("channel", "101")

        urls[camera_id] = (
            f"rtsp://{username}:{password}@{ip}:{port}"
            f"/Streaming/Channels/{channel}"
        )

    return urls

def main():
    global ACTIVE_DOOR, SELECTING_LINE
    global PENDING_POINTS
    global INPUT_MODE, INPUT_BUFFER
    global CANVAS_WIDTH, CANVAS_HEIGHT

    try:
        urls = load_camera_urls()
        print("[CAMERA] Konfigurasi CCTV berhasil dimuat.")

    except Exception as error:
        print("[ERROR] Gagal membaca konfigurasi CCTV:", error)
        return

    print("=" * 60)
    print("LOBBY PEOPLE TRACKING - STABLE (DUAL CCTV)")
    print(f"{len(CAMERAS)} CCTV | {len(DOORS)} PINTU | YOLO + ByteTrack")
    print(f"YOLO IMGSZ: {IMGSZ} | FPS maks per kamera: {INFERENCE_MAX_FPS}")
    print(f"Tracker: {TRACKER_CONFIG}")
    print("=" * 60)

    if not Path(TRACKER_CONFIG).exists():
        print(f"[ERROR] File tracker tidak ditemukan: {TRACKER_CONFIG}")
        print(
            "Simpan custom_bytetrack_stable.yaml "
            "di folder yang sama dengan script."
        )
        return

    # Satu instance model per kamera supaya tracker tidak tercampur.
    models = {}
    inference_device = None

    try:
        for index, camera_id in enumerate(CAMERAS):
            if index == 0:
                model, inference_device = load_model()
            else:
                model, _ = load_model(
                    device_hint=inference_device,
                    use_openvino=inference_device is not None,
                )

            models[camera_id] = model

    except Exception as error:
        print("[ERROR] Gagal memuat model YOLO:", error)
        return

    cameras = {
        camera_id: CameraState(camera_id, urls[camera_id])
        for camera_id in CAMERAS
    }

    # --------------------------------------------------------
    # JALANKAN THREAD
    # --------------------------------------------------------

    log_thread = threading.Thread(target=log_writer_worker, daemon=True)
    log_thread.start()

    threads = []

    for camera_id, cam in cameras.items():
        threads.append(
            threading.Thread(target=capture_worker, args=(cam,), daemon=True)
        )
        threads.append(
            threading.Thread(
                target=inference_worker,
                args=(cam, models[camera_id], inference_device),
                daemon=True,
            )
        )

    for thread in threads:
        thread.start()

    # Tunggu frame pertama dari semua CCTV.
    wait_start = time.monotonic()

    while time.monotonic() - wait_start < FIRST_FRAME_TIMEOUT:
        if all(cam.frame is not None for cam in cameras.values()):
            break

        time.sleep(0.1)

    missing_cameras = [
        CAMERAS[camera_id]["title"]
        for camera_id, cam in cameras.items()
        if cam.frame is None
    ]

    if missing_cameras:
        STOP_EVENT.set()
        LOG_STOP.set()
        print(
            f"[ERROR] CCTV tidak mengirim frame dalam "
            f"{FIRST_FRAME_TIMEOUT:.0f} detik: {', '.join(missing_cameras)}"
        )
        return

    # Ukuran kerja tiap kamera + tata letak panel dalam grid (COLUMNS kolom).
    sizes = {}

    for camera_id, cam in cameras.items():
        with cam.lock:
            frame_height, frame_width = cam.frame.shape[:2]

        FRAME_SIZES[camera_id] = (frame_width, frame_height)

        scale = min(
            PANEL_WIDTH / frame_width,
            PANEL_HEIGHT / frame_height,
            1.0,
        )

        sizes[camera_id] = (
            int(frame_width * scale),
            int(frame_height * scale),
        )

    cell_w = max(width for width, _ in sizes.values())
    cell_h = max(height for _, height in sizes.values())

    for index, camera_id in enumerate(cameras):
        view_width, view_height = sizes[camera_id]

        VIEWS[camera_id] = {
            "x": (index % COLUMNS) * cell_w,
            "y": (index // COLUMNS) * cell_h,
            "w": view_width,
            "h": view_height,
        }

    CANVAS_WIDTH = cell_w * min(COLUMNS, len(cameras))
    CANVAS_HEIGHT = cell_h * math.ceil(len(cameras) / COLUMNS)

    load_config()
    load_state()

    last_state_save = time.monotonic()
    last_reset_date = datetime.now().strftime("%Y-%m-%d")

    cv2.namedWindow(WINDOW_NAME, cv2.WINDOW_NORMAL)
    cv2.resizeWindow(WINDOW_NAME, CANVAS_WIDTH, CANVAS_HEIGHT)
    cv2.setMouseCallback(WINDOW_NAME, mouse_callback)

    print()
    print("KONTROL:")
    print("1 = Atur garis PINTU KELUAR       (panel LANTAI 1)")
    print("2 = Atur garis AREA PRODUKSI      (panel LANTAI 1)")
    print("3 = Atur garis OFFICE LANTAI 1    (panel LANTAI 1)")
    print("4 = Atur garis TANGGA             (panel LANTAI 2)")
    print("5 = Atur garis OFFICE LANTAI 2    (panel LANTAI 2)")
    print("6 = Atur garis RUANG MEETING 2  (panel CCTV LANTAI 2)")
    print("7 = Atur garis RUANG MEETING 1  (panel CCTV2 LANTAI 2)")
    print("R = Balik arah IN/OUT pintu terpilih")
    print("S = Simpan konfigurasi garis")
    print("+ / - = Koreksi okupansi Lobby (+1 / -1)")
    print("K = Isi okupansi Lobby manual (ketik angka di jendela, Enter simpan, Esc batal)")
    print("X = Reset hitungan crossing dan okupansi area")
    print("Q = Keluar")
    print("Klik jendela CCTV terlebih dahulu sebelum menekan tombol.")
    print("Garis tangga: IN = naik (Lobby -> Lantai 2), OUT = turun (Lantai 2 -> Lobby).")
    print()

    # --------------------------------------------------------
    # LOOP TAMPILAN
    # --------------------------------------------------------

    # Panel digambar ulang hanya kalau isinya berubah (hasil AI baru,
    # hitungan berubah, garis diubah, dsb.). Selain itu canvas lama dipakai.
    panel_cache = {}
    canvas = None

    try:
        while True:
            loop_start = time.perf_counter()

            # Auto-reset harian dan simpan state berkala.
            now_dt = datetime.now()
            today = now_dt.strftime("%Y-%m-%d")

            if (
                AUTO_RESET_HOUR is not None
                and today != last_reset_date
                and now_dt.hour >= AUTO_RESET_HOUR
            ):
                reset_counts("auto-reset harian")
                last_reset_date = today
                last_state_save = time.monotonic()

            if time.monotonic() - last_state_save > STATE_SAVE_INTERVAL:
                request_save()
                last_state_save = time.monotonic()

            snapshot = get_snapshot()
            state_key = make_state_key(snapshot)

            selecting_key = (
                (ACTIVE_DOOR, tuple(PENDING_POINTS)) if SELECTING_LINE else None
            )

            fps_text = "/".join(
                f"{cameras[camera_id].fps:.0f}" for camera_id in CAMERAS
            )

            changed = canvas is None

            for camera_id, cam in cameras.items():
                view_data = cam.get_view_data()

                key = (
                    view_data[0],
                    state_key,
                    selecting_key,
                    fps_text,
                    cam.camera_ok,
                    cam.last_error,
                )

                cached = panel_cache.get(camera_id)

                if cached is not None and cached[0] == key:
                    continue

                panel = render_camera(cam, view_data, fps_text, snapshot)
                view = VIEWS[camera_id]

                if panel.shape[1] != view["w"] or panel.shape[0] != view["h"]:
                    panel = cv2.resize(
                        panel,
                        (view["w"], view["h"]),
                        interpolation=cv2.INTER_AREA,
                    )

                panel_cache[camera_id] = (key, panel)
                changed = True

            if changed:
                canvas = np.zeros(
                    (CANVAS_HEIGHT, CANVAS_WIDTH, 3), dtype="uint8"
                )

                for camera_id in cameras:
                    view = VIEWS[camera_id]
                    panel = panel_cache[camera_id][1]

                    canvas[
                        view["y"]:view["y"] + view["h"],
                        view["x"]:view["x"] + view["w"],
                    ] = panel

                    # Garis pemisah antar panel.
                    if view["x"] > 0:
                        cv2.line(
                            canvas,
                            (view["x"], view["y"]),
                            (view["x"], view["y"] + view["h"]),
                            (90, 90, 90),
                            1,
                        )

                    if view["y"] > 0:
                        cv2.line(
                            canvas,
                            (view["x"], view["y"]),
                            (view["x"] + view["w"], view["y"]),
                            (90, 90, 90),
                            1,
                        )

            if INPUT_MODE:
                shown = canvas.copy()
                draw_input_prompt(shown)
            else:
                shown = canvas

            cv2.imshow(WINDOW_NAME, shown)

            # Batasi FPS tampilan supaya loop tidak berputar tanpa henti.
            spare = 1.0 / UI_FPS - (time.perf_counter() - loop_start)
            key = cv2.waitKey(max(1, int(spare * 1000))) & 0xFF

            # Jendela ditutup lewat tombol X.
            if cv2.getWindowProperty(WINDOW_NAME, cv2.WND_PROP_VISIBLE) < 1:
                break

            if INPUT_MODE:
                # Selama mode input, semua tombol masuk ke kotak ketik.
                if ord("0") <= key <= ord("9"):
                    if len(INPUT_BUFFER) < 3:
                        INPUT_BUFFER += chr(key)

                elif key in (8, 127):  # Backspace
                    INPUT_BUFFER = INPUT_BUFFER[:-1]

                elif key in (13, 10):  # Enter
                    if INPUT_BUFFER:
                        set_area("Lobby", int(INPUT_BUFFER))
                    else:
                        print("[INFO] Input kosong, dibatalkan.")

                    INPUT_MODE = False
                    INPUT_BUFFER = ""

                elif key == 27:  # Esc
                    INPUT_MODE = False
                    INPUT_BUFFER = ""
                    print("[INFO] Input okupansi dibatalkan.")

            elif key in (
                ord("1"), ord("2"), ord("3"), ord("4"),
                ord("5"), ord("6"), ord("7")
            ):
                ACTIVE_DOOR = int(chr(key))
                SELECTING_LINE = True
                PENDING_POINTS = []

                print(
                    f"[CONFIG] Pilih dua titik untuk "
                    f"{DOORS[ACTIVE_DOOR]['name']} di panel "
                    f"{CAMERAS[DOORS[ACTIVE_DOOR]['camera']]['title']}."
                )

            elif key in (ord("r"), ord("R")):
                with STATE_LOCK:
                    DOORS[ACTIVE_DOOR]["reverse"] = (
                        not DOORS[ACTIVE_DOOR]["reverse"]
                    )

                    TRACK_SIDES[ACTIVE_DOOR].clear()
                    LAST_FOOT_POINT[ACTIVE_DOOR].clear()

                save_config()

                print(f"[CONFIG] Arah {DOORS[ACTIVE_DOOR]['name']} dibalik.")

            elif key in (ord("s"), ord("S")):
                save_config()

            elif key in (ord("+"), ord("=")):
                adjust_area("Lobby", +1)

            elif key == ord("-"):
                adjust_area("Lobby", -1)

            elif key in (ord("k"), ord("K")):
                INPUT_MODE = True
                INPUT_BUFFER = ""
                SELECTING_LINE = False
                PENDING_POINTS = []

            elif key in (ord("x"), ord("X")):
                reset_counts("manual")

            elif key in (ord("q"), ord("Q")):
                break

    except KeyboardInterrupt:
        print("[INFO] Program dihentikan dari keyboard.")

    except Exception as error:
        print("[FATAL ERROR]", repr(error))

    finally:
        STOP_EVENT.set()

        for thread in threads:
            thread.join(timeout=3)

        # Thread penulis berhenti terakhir supaya sisa log sempat ditulis.
        LOG_STOP.set()
        log_thread.join(timeout=5)

        # Simpan state terakhir supaya restart tidak mulai dari 0.
        save_state()

        cv2.destroyAllWindows()

        print("[INFO] People tracking dihentikan.")
